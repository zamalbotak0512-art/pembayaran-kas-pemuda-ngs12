const $=id=>document.getElementById(id);const rupiah=n=>new Intl.NumberFormat('id-ID',{style:'currency',currency:'IDR',maximumFractionDigits:0}).format(n||0);
let tx=JSON.parse(localStorage.getItem('neglasari_tx')||'[]');let qris=localStorage.getItem('neglasari_qris')||'';let pin=localStorage.getItem('neglasari_pin')||'123456';let pendingQris='';
function save(){localStorage.setItem('neglasari_tx',JSON.stringify(tx));render()}
function showView(v){$('memberView').classList.toggle('hidden',v!=='member');$('treasurerView').classList.toggle('hidden',v!=='treasurer');$('memberTab').classList.toggle('active',v==='member');$('treasurerTab').classList.toggle('active',v==='treasurer')}
function unlock(){if($('pinInput').value===pin){$('pinCard').classList.add('hidden');$('dashboard').classList.remove('hidden');$('pinError').textContent='';$('pinInput').value='';render()}else $('pinError').textContent='PIN salah. Silakan coba lagi.'}
function lock(){$('dashboard').classList.add('hidden');$('pinCard').classList.remove('hidden')}
function addTransaction(){let name=$('person').value.trim(),amount=Number($('amount').value),desc=$('description').value.trim();if(!name||!Number.isFinite(amount)||amount<=0){alert('Isi nama anggota dan jumlah yang benar.');return}tx.unshift({id:Date.now(),date:new Date().toLocaleDateString('id-ID'),name,amount,desc:desc||'Iuran kas'});save();$('person').value='';$('amount').value='';$('description').value=''}
function removeTransaction(id){if(confirm('Hapus transaksi ini?')){tx=tx.filter(t=>t.id!==id);save()}}
function clearTransactions(){if(confirm('Yakin menghapus semua transaksi?')){tx=[];save()}}
function changePin(){if($('oldPin').value!==pin){$('pinMessage').textContent='PIN saat ini tidak sesuai.';return}let p=$('newPin').value;if(!/^\d{4,12}$/.test(p)){$('pinMessage').textContent='PIN baru harus terdiri dari 4–12 angka.';return}pin=p;localStorage.setItem('neglasari_pin',pin);$('oldPin').value='';$('newPin').value='';$('pinMessage').textContent='PIN berhasil diganti di perangkat ini.'}

function saveQris(){if(!pendingQris){alert('Pilih gambar QRIS terlebih dahulu.');return}qris=pendingQris;localStorage.setItem('neglasari_qris',qris);pendingQris='';render();$('qrisPreview').innerHTML='';$('qrisFile').value='';alert('QRIS tersimpan di perangkat ini.')}
